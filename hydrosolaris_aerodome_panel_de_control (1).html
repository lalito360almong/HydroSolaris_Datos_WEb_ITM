Carga este código en el Arduino Cerebro y cierra el Monitor Serial del IDE antes de conectar por USB. 

/*
 * Proyecto: HydroSolaris: AeroDome
 * Institución: Instituto Tecnológico México - 5to A Mecatrónica
 * Módulo: Arduino Cerebro (Lectura de 10 Señales Digitales -> Transmisión JSON Serial)
 * 
 * Función: Lee exclusivamente los 5 pines digitales del Nodo 1 y los 5 pines
 * digitales del Nodo 3, y los transmite en formato JSON a 9600 baudios.
 */

// --- Pines de Entrada desde NODO 1 (Iluminación, Radiación y Clima Interno) ---
const int pinIn_N1_Noche     = 2;  // Recibe Pin D3 de Nodo 1 (HIGH: Noche / LOW: Día)
const int pinIn_N1_Radiacion = 3;  // Recibe Pin D4 de Nodo 1 (HIGH: Radiación Baja / LOW: Normal)
const int pinIn_N1_Humedad   = 4;  // Recibe Pin D5 de Nodo 1 (HIGH: Fuera de rango / LOW: Estable)
const int pinIn_N1_Temp      = 5;  // Recibe Pin D6 de Nodo 1 (HIGH: >35°C / LOW: Segura)
const int pinIn_N1_LuzManual = 6;  // Recibe Pin D7 de Nodo 1 (HIGH: Falta de sol prolongada / LOW: Inactiva)

// --- Pines de Entrada desde NODO 3 (Ciclo Hídrico y Nutrición) ---
const int pinIn_N3_Lluvia    = 7;  // Recibe Pin D4 de Nodo 3 (HIGH: Lluvia activa / LOW: Seco)
const int pinIn_N3_SueloSeco = 8;  // Recibe Pin D5 de Nodo 3 (HIGH: Suelo seco - Motor C ON / LOW: Óptima)
const int pinIn_N3_Tanque100 = 9;  // Recibe Pin D6 de Nodo 3 (HIGH: Tanque lleno 100% / LOW: Disponible)
const int pinIn_N3_Tanque60  = 10; // Recibe Pin D10 de Nodo 3 (HIGH: Nivel <=60% / LOW: >60%)
const int pinIn_N3_MotoresAB = 11; // Recibe Pin D11 de Nodo 3 (HIGH: Succión pluvial activa / LOW: Apagados)

void setup() {
  Serial.begin(9600);

  pinMode(pinIn_N1_Noche, INPUT);
  pinMode(pinIn_N1_Radiacion, INPUT);
  pinMode(pinIn_N1_Humedad, INPUT);
  pinMode(pinIn_N1_Temp, INPUT);
  pinMode(pinIn_N1_LuzManual, INPUT);

  pinMode(pinIn_N3_Lluvia, INPUT);
  pinMode(pinIn_N3_SueloSeco, INPUT);
  pinMode(pinIn_N3_Tanque100, INPUT);
  pinMode(pinIn_N3_Tanque60, INPUT);
  pinMode(pinIn_N3_MotoresAB, INPUT);
}

void loop() {
  int n1_noche  = (digitalRead(pinIn_N1_Noche) == HIGH) ? 1 : 0;
  int n1_rad    = (digitalRead(pinIn_N1_Radiacion) == HIGH) ? 1 : 0;
  int n1_hum    = (digitalRead(pinIn_N1_Humedad) == HIGH) ? 1 : 0;
  int n1_temp   = (digitalRead(pinIn_N1_Temp) == HIGH) ? 1 : 0;
  int n1_luz    = (digitalRead(pinIn_N1_LuzManual) == HIGH) ? 1 : 0;

  int n3_lluvia = (digitalRead(pinIn_N3_Lluvia) == HIGH) ? 1 : 0;
  int n3_suelo  = (digitalRead(pinIn_N3_SueloSeco) == HIGH) ? 1 : 0;
  int n3_lleno  = (digitalRead(pinIn_N3_Tanque100) == HIGH) ? 1 : 0;
  int n3_h60    = (digitalRead(pinIn_N3_Tanque60) == HIGH) ? 1 : 0;
  int n3_motAB  = (digitalRead(pinIn_N3_MotoresAB) == HIGH) ? 1 : 0;

  Serial.print("{\"n1_noche\":");  Serial.print(n1_noche);
  Serial.print(",\"n1_rad\":");    Serial.print(n1_rad);
  Serial.print(",\"n1_hum\":");    Serial.print(n1_hum);
  Serial.print(",\"n1_temp\":");   Serial.print(n1_temp);
  Serial.print(",\"n1_luz\":");    Serial.print(n1_luz);
  Serial.print(",\"n3_lluvia\":"); Serial.print(n3_lluvia);
  Serial.print(",\"n3_suelo\":");  Serial.print(n3_suelo);
  Serial.print(",\"n3_lleno\":");  Serial.print(n3_lleno);
  Serial.print(",\"n3_h60\":");    Serial.print(n3_h60);
  Serial.print(",\"n3_motAB\":");  Serial.print(n3_motAB);
  Serial.println("}");

  delay(1000);
}
